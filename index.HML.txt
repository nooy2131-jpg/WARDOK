        // Open Mobile Share & QR Code Modal
        function openMobileLinkModal() {
            const currentUrl = window.location.href;
            const urlInput = document.getElementById('mobile-share-url');
            const warningBox = document.getElementById('preview-url-warning');

            // Detect if running inside a preview iframe / local session
            const isPreviewEnv = currentUrl.startsWith('blob:') || 
                                 currentUrl.includes('googleusercontent.com') || 
                                 currentUrl.includes('localhost') || 
                                 currentUrl.startsWith('about:');

            if (warningBox) {
                if (isPreviewEnv) {
                    warningBox.classList.remove('hidden');
                } else {
                    warningBox.classList.add('hidden');
                }
            }

            if (urlInput) {
                if (!urlInput.value || urlInput.value === '') {
                    urlInput.value = isPreviewEnv ? '' : currentUrl;
                }
            }

            updateQrCodeFromInput();
            document.getElementById('modal-mobile-share').classList.remove('hidden');
        }

        // Update QR Code dynamically based on input URL
        function updateQrCodeFromInput() {
            const urlInput = document.getElementById('mobile-share-url');
            const qrImg = document.getElementById('mobile-qr-img');
            const statusText = document.getElementById('qr-status-text');

            let targetUrl = urlInput ? urlInput.value.trim() : '';

            if (!targetUrl) {
                targetUrl = window.location.href;
            }

            if (qrImg) {
                qrImg.src = `https://api.qrserver.com/v1/create-qr-code/?size=240x240&data=${encodeURIComponent(targetUrl)}`;
            }

            if (statusText) {
                statusText.innerText = targetUrl ? `QR Code สำหรับ: ${targetUrl.substring(0, 35)}${targetUrl.length > 35 ? '...' : ''}` : 'กรุณากรอก URL เพื่อสร้าง QR Code';
            }
        }

        // Copy Mobile Link to Clipboard
        function copyMobileLink() {
            const urlInput = document.getElementById('mobile-share-url');
            if (!urlInput || !urlInput.value) {
                showToast('กรุณากรอก URL เว็บไซต์ก่อนคัดลอก', true);
                return;
            }

            urlInput.select();
            urlInput.setSelectionRange(0, 99999);
            
            try {
                document.execCommand('copy');
                showToast('คัดลอก Link เรียบร้อยแล้ว!');
            } catch (err) {
                showToast('ไม่สามารถคัดลอกอัตโนมัติได้ กรุณาเลือกและคัดลอกด้วยตนเอง', true);
            }
        }

        // Initialize Sample Data if local storage is empty
        function initSampleData() {
            const saved = localStorage.getItem('ward_patient_records');
            if (saved) {
                try {
                    records = JSON.parse(saved);
                } catch(e) {
                    records = generateMockRecords();
                }
            } else {
                records = generateMockRecords();
                saveRecords();
            }
        }

        // Generate 7 Days of Mock Data for initial interactive demo
        function generateMockRecords() {
            const shifts = ['เช้า', 'บ่าย', 'ดึก'];
            const mock = [];
            const today = new Date();

            for (let i = 6; i >= 0; i--) {
                const d = new Date(today);
                d.setDate(d.getDate() - i);
                const dateStr = d.toISOString().split('T')[0];

                shifts.forEach(shift => {
                    const totalBeds = 30;
                    const newAdmit = Math.floor(Math.random() * 4);
                    const discharge = Math.floor(Math.random() * 3);
                    const refer = Math.floor(Math.random() * 2);
                    
                    // Generate logical breakdown
                    const t1 = 5 + Math.floor(Math.random() * 4);
                    const t2 = 6 + Math.floor(Math.random() * 4);
                    const t3 = 4 + Math.floor(Math.random() * 3);
                    const t4 = 2 + Math.floor(Math.random() * 2);
                    const t5 = Math.floor(Math.random() * 2);

                    const currentPatients = t1 + t2 + t3 + t4 + t5;
                    const availableBeds = totalBeds - currentPatients;

                    mock.push({
                        id: 'REC_' + Date.now() + '_' + Math.random().toString(36).substr(2, 5),
                        date: dateStr,
                        shift: shift,
                        totalBeds: totalBeds,
                        currentPatients: currentPatients,
                        availableBeds: availableBeds,
                        newAdmissions: newAdmit,
                        discharges: discharge,
                        refers: refer,
                        type1: t1,
                        type2: t2,
                        type3: t3,
                        type4: t4,
                        type5: t5,
                        note: 'รายงานปกติประจำเวร' + shift,
                        timestamp: d.getTime()
                    });
                });
            }
            return mock;
        }

        // Auto Calculate Available Beds
        function autoCalculateBeds() {
            const capacity = parseInt(document.getElementById('input-totalBeds').value) || DEFAULT_BED_CAPACITY;
            const current = parseInt(document.getElementById('input-currentPatients').value) || 0;
            const available = Math.max(0, capacity - current);
            document.getElementById('input-availableBeds').value = available;
        }

        // Validate that Patient Types 1-5 sum equals Total Current Patients
        function validateTypesSum() {
            const current = parseInt(document.getElementById('input-currentPatients').value) || 0;
            const t1 = parseInt(document.getElementById('input-type1').value) || 0;
            const t2 = parseInt(document.getElementById('input-type2').value) || 0;
            const t3 = parseInt(document.getElementById('input-type3').value) || 0;
            const t4 = parseInt(document.getElementById('input-type4').value) || 0;
            const t5 = parseInt(document.getElementById('input-type5').value) || 0;

            const sum = t1 + t2 + t3 + t4 + t5;
            const badge = document.getElementById('types-sum-badge');
            const sumVal = document.getElementById('types-sum-val');
            const targetVal = document.getElementById('types-target-val');

            sumVal.innerText = sum;
            targetVal.innerText = current;

            if (sum === current) {
                badge.className = "text-xs px-3 py-1 rounded-full font-medium bg-emerald-100 text-emerald-800 border border-emerald-300";
            } else {
                badge.className = "text-xs px-3 py-1 rounded-full font-medium bg-rose-100 text-rose-800 border border-rose-300 animate-pulse";
            }
        }

        // Handle Form Submission
        function handleFormSubmit(e) {
            e.preventDefault();

            const date = document.getElementById('input-date').value;
            const shift = document.getElementById('input-shift').value;
            const totalBeds = parseInt(document.getElementById('input-totalBeds').value) || DEFAULT_BED_CAPACITY;
            const currentPatients = parseInt(document.getElementById('input-currentPatients').value) || 0;
            const availableBeds = Math.max(0, totalBeds - currentPatients);
            const newAdmissions = parseInt(document.getElementById('input-newAdmissions').value) || 0;
            const discharges = parseInt(document.getElementById('input-discharges').value) || 0;
            const refers = parseInt(document.getElementById('input-refers').value) || 0;

            const t1 = parseInt(document.getElementById('input-type1').value) || 0;
            const t2 = parseInt(document.getElementById('input-type2').value) || 0;
            const t3 = parseInt(document.getElementById('input-type3').value) || 0;
            const t4 = parseInt(document.getElementById('input-type4').value) || 0;
            const t5 = parseInt(document.getElementById('input-type5').value) || 0;
            const note = document.getElementById('input-note').value;

            // Warning validation check
            if ((t1 + t2 + t3 + t4 + t5) !== currentPatients) {
                showToast(`คำเตือน: ผลรวมประเภทผู้ป่วย 1-5 (${t1+t2+t3+t4+t5}) ไม่ตรงกับผู้ป่วยรวม (${currentPatients})`, true);
            }

            const editId = document.getElementById('form-id').value;

            if (editId) {
                // Update Existing Record
                const idx = records.findIndex(r => r.id === editId);
                if (idx !== -1) {
                    records[idx] = {
                        ...records[idx],
                        date, shift, totalBeds, currentPatients, availableBeds,
                        newAdmissions, discharges, refers,
                        type1: t1, type2: t2, type3: t3, type4: t4, type5: t5,
                        note
                    };
                    showToast('อัปเดตข้อมูลการรายงานสำเร็จ');
                }
            } else {
                // Check if date + shift already exists
                const existingIdx = records.findIndex(r => r.date === date && r.shift === shift);
                if (existingIdx !== -1) {
                    // Overwrite or Alert
                    records[existingIdx] = {
                        id: records[existingIdx].id,
                        date, shift, totalBeds, currentPatients, availableBeds,
                        newAdmissions, discharges, refers,
                        type1: t1, type2: t2, type3: t3, type4: t4, type5: t5,
                        note, timestamp: Date.now()
                    };
                    showToast(`อัปเดตทับข้อมูลวันที่ ${date} เวร${shift} เรียบร้อย`);
                } else {
                    // New Entry
                    const newRecord = {
                        id: 'REC_' + Date.now(),
                        date, shift, totalBeds, currentPatients, availableBeds,
                        newAdmissions, discharges, refers,
                        type1: t1, type2: t2, type3: t3, type4: t4, type5: t5,
                        note, timestamp: Date.now()
                    };
                    records.push(newRecord);
                    showToast('บันทึกข้อมูลเวรใหม่เรียบร้อย');
                }
            }

            saveRecords();
            clearForm();
            switchTab('dashboard');
        }

        // Render Dashboard Calculations & Charts
        function renderDashboard() {
            if (records.length === 0) return;

            // Sort records chronologically
            const sorted = [...records].sort((a,b) => (a.date + a.shift).localeCompare(b.date + b.shift));

            // Selected Filters
            const filterDate = document.getElementById('filter-date').value;
            const filterShift = document.getElementById('filter-shift').value;

            // Filtered records
            let filtered = sorted;
            if (filterDate) {
                filtered = filtered.filter(r => r.date === filterDate);
            }

            let displayRecord = null;

            if (filtered.length > 0) {
                if (filterShift === 'latest') {
                    displayRecord = filtered[filtered.length - 1];
                } else if (filterShift === 'all') {
                    // Aggregate multiple shift records
                    displayRecord = aggregateRecords(filtered);
                } else {
                    const matched = filtered.filter(r => r.shift === filterShift);
                    displayRecord = matched.length > 0 ? matched[matched.length - 1] : aggregateRecords(filtered);
                }
            } else {
                // Fallback if no exact date match: use overall latest record
                displayRecord = sorted[sorted.length - 1];
            }

            if (!displayRecord) return;

            // Update KPI Cards
            document.getElementById('kpi-current').innerText = displayRecord.currentPatients || 0;
            document.getElementById('kpi-available').innerText = displayRecord.availableBeds || 0;
            document.getElementById('kpi-admit').innerText = displayRecord.newAdmissions || 0;
            document.getElementById('kpi-discharge').innerText = displayRecord.discharges || 0;
            document.getElementById('kpi-refer').innerText = displayRecord.refers || 0;

            const totalCap = displayRecord.totalBeds || DEFAULT_BED_CAPACITY;
            const occRate = Math.round(((displayRecord.currentPatients || 0) / totalCap) * 100);
            
            document.getElementById('kpi-occupancy-rate').innerText = `อัตราครองเตียง: ${occRate}%`;
            document.getElementById('kpi-total-capacity').innerText = `ความจุทั้งหมด: ${totalCap} เตียง`;
            
            document.getElementById('kpi-occupancy-bar').style.width = `${Math.min(100, occRate)}%`;
            document.getElementById('kpi-available-bar').style.width = `${Math.min(100, 100 - occRate)}%`;

            // Patient Levels
            document.getElementById('kpi-type-1').innerText = displayRecord.type1 || 0;
            document.getElementById('kpi-type-2').innerText = displayRecord.type2 || 0;
            document.getElementById('kpi-type-3').innerText = displayRecord.type3 || 0;
            document.getElementById('kpi-type-4').innerText = displayRecord.type4 || 0;
            document.getElementById('kpi-type-5').innerText = displayRecord.type5 || 0;

            document.getElementById('last-updated-text').innerText = `ข้อมูล ณ วันที่: ${displayRecord.date} | เวร: ${displayRecord.shift}`;

            // Render Charts
            renderCharts(sorted, displayRecord);
            renderShiftSummary(sorted);
        }

        // Aggregate Multiple Records into one KPI view
        function aggregateRecords(list) {
            if (list.length === 0) return null;
            const last = list[list.length - 1];
            return {
                date: last.date,
                shift: 'รวมตลอดวัน',
                totalBeds: last.totalBeds,
                currentPatients: last.currentPatients,
                availableBeds: last.availableBeds,
                newAdmissions: list.reduce((s, r) => s + r.newAdmissions, 0),
                discharges: list.reduce((s, r) => s + r.discharges, 0),
                refers: list.reduce((s, r) => s + r.refers, 0),
                type1: last.type1,
                type2: last.type2,
                type3: last.type3,
                type4: last.type4,
                type5: last.type5,
            };
        }

        // Render Shift Workload Summary
        function renderShiftSummary(sortedRecords) {
            const container = document.getElementById('shift-summary-cards');
            const shifts = ['เช้า', 'บ่าย', 'ดึก'];
            const colors = {
                'เช้า': 'bg-blue-50 border-blue-200 text-blue-800',
                'บ่าย': 'bg-amber-50 border-amber-200 text-amber-800',
                'ดึก': 'bg-indigo-50 border-indigo-200 text-indigo-800'
            };

            let html = '';
            shifts.forEach(s => {
                const shiftRecords = sortedRecords.filter(r => r.shift === s);
                const count = shiftRecords.length || 1;
                const avgAdmit = (shiftRecords.reduce((a, b) => a + b.newAdmissions, 0) / count).toFixed(1);
                const avgDischarge = (shiftRecords.reduce((a, b) => a + b.discharges, 0) / count).toFixed(1);

                html += `
                    <div class="p-3 rounded-xl border ${colors[s]} flex justify-between items-center">
                        <div>
                            <span class="font-bold text-xs uppercase tracking-wider block">เวร${s}</span>
                            <span class="text-[11px] text-slate-600">รับใหม่เฉลี่ย: ${avgAdmit} คน/เวร</span>
                        </div>
                        <div class="text-right">
                            <span class="text-xs font-semibold block text-slate-700">จำหน่ายเฉลี่ย</span>
                            <span class="text-xs font-bold">${avgDischarge} คน</span>
                        </div>
                    </div>
                `;
            });
            container.innerHTML = html;
        }

        // Export Data Records to CSV
        function exportToCSV() {
            if (records.length === 0) {
                showToast('ไม่มีข้อมูลสำหรับส่งออก', true);
                return;
            }

            const headers = ['Date', 'Shift', 'TotalBeds', 'CurrentPatients', 'AvailableBeds', 'NewAdmissions', 'Discharges', 'Refers', 'Type1', 'Type2', 'Type3', 'Type4', 'Type5', 'Note'];
            const rows = records.map(r => [
                r.date,
                `"${r.shift}"`,
                r.totalBeds,
                r.currentPatients,
                r.availableBeds,
                r.newAdmissions,
                r.discharges,
                r.refers,
                r.type1,
                r.type2,
                r.type3,
                r.type4,
                r.type5,
                `"${(r.note || '').replace(/"/g, '""')}"`
            ]);

            const csvContent = "\uFEFF" + [headers.join(','), ...rows.map(e => e.join(','))].join('\n');
            const blob = new Blob([csvContent], { type: 'text/csv;charset=utf-8;' });
            const url = URL.createObjectURL(blob);
            
            const a = document.createElement('a');
            a.href = url;
            a.download = `patient_report_${new Date().toISOString().split('T')[0]}.csv`;
            a.click();
            URL.revokeObjectURL(url);

            showToast('ส่งออกไฟล์ CSV สำเร็จ');
        }
